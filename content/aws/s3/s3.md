##### Cli で VPC Interface endpoint から S3 へアクセス

```shell
aws s3 --endpoint-url https://bucket.ENDPOINTURL.s3.ap-northeast-1.vpce.amazonaws.com COMMAND s3://BUCKETNAME
```

##### Cli で S3 のファイル名一覧を再帰的に取得する

```shell
aws s3 ls s3://your-bucket-name/ --recursive | awk '{print $4}'
```

##### Cli で S3 のファイル名一覧とサイズ（MB）を再帰的に取得する

```shell
aws s3 ls s3://your-bucket-name/ --recursive | awk '{print $4, $3 / 1024 / 1024 " MB"}'
aws s3 ls s3://your-bucket-name/ --recursive --human --sum
```

##### Cli で S3 のファイル数をカウントする

```shell
aws s3 ls s3://parquet-8mbfile-output/ --recursive | wc -l
```

##### Cli で S3 の配下に特定のパターンのファイルを削除する

```shell
aws s3 ls s3://your-bucket-name/path/to/folder/ --recursive | grep 'run-1699431295238-part-r-.*' | awk '{print $4}' | while read file; do
   aws s3 rm s3://your-bucket-name/$file
done
```

##### Cli で S3 へファイルアップロード（Powershell）

```powershell
# マルチパートアップロード設定 - スレッドの個数
aws configure set default.s3.max_concurrent_requests 100

# マルチパートアップロード設定 - マルチパートアップロードを採用するファイルサイズの閾値
aws configure set default.s3.multipart_threshold 512MB

# マルチパートアップロード設定 - チャンクサイズ
aws configure set default.s3.multipart_chunksize 32MB

# マルチパートアップロード設定 - 最大帯域幅
aws configure set default.s3.max_bandwidth 200MB/s

$folderPath = 'C:\Users\Administrator\Downloads\S3' # Windowsローカルのフォルダパス（ここではS3配下にyyyymmdd存在する想定）
$bucketName = '${BUCKET_NAME}' # バケット名
$prefixName = '${S3_PREFIX_URI}' # S3 Prefix

$dirs = Get-ChildItem -Path $folderPath -Directory

$jobs = foreach ($dir in $dirs) {
    # マルチスレッド処理
    Start-Job -ScriptBlock {
        param($dir, $bucketName, $prefixName, $folderPath)

        # フォルダ名変換処理
        $folderName = $dir.Name

        $year = $folderName.Substring(0, 4)
        $month = $folderName.Substring(4, 2)
        $day = $folderName.Substring(6, 2)
        $hour = $folderName.Substring(8, 2)

        $convertedName = "year=$year/month=$month/day=$day/hour=$hour"

        Set-Location -Path $folderPath
        # Assume aws is available in the PATH
        aws s3 cp $folderName s3://$bucketName/$prefixName/$convertedName/ --recursive

    } -ArgumentList $dir, $bucketName, $prefixName, $folderPath
}

# Wait for all jobs to complete
$jobs | Wait-Job

# Get the output from all jobs
$jobs | Receive-Job

# Clean up the jobs
$jobs | Remove-Job
```

##### Cli で S3 のファイルを結合する

```shell
#!bin/bash

(
aws s3 cp s3://BUCKET_NAME/PREFIX/.../FILENAME_1.gz -;
aws s3 cp s3://BUCKET_NAME/PREFIX/.../FILENAME_2.gz -;

) | aws s3 cp - s3://NEW_BUCKET_NAME/PREFIX/.../FILENAME.gz -;
```

##### Python で S3 を操作する

```shell
# s3 - boto3 client init
client = boto3.client('s3')

# s3 - check if bucket exist
s3Client.head_bucket(Bucket=bucket)

# s3 - create bucket
# 貼り付ける際はインデントに注意してください
    if current_region == 'us-east-1':
        response = s3Client.create_bucket(Bucket=name)
    else:
        response = s3Client.create_bucket(
          Bucket=name,
          CreateBucketConfiguration={
              'LocationConstraint': current_region
          })

# s3 - wait for bucket create
# 貼り付ける際はインデントに注意してください
    waiter = s3Client.get_waiter('bucket_exists')
    waiter.wait(Bucket=bucket)

# s3 - upload file
# 貼り付ける際はインデントに注意してください
    response = s3Client.upload_file(
        Bucket=bucket,
        Key=key,
        Filename=name,
        ExtraArgs={
            'ContentType': contentType,
            'Metadata': metadata
            }
    )

# s3 - download file on memory
# 貼り付ける際はインデントに注意してください
    s3Client.download_fileobj(
      Bucket=bucket,
      Key=key,
      Fileobj=bytes_buffer)

# s3 - upload object data
# 貼り付ける際はインデントに注意してください
    s3Client.put_object(
        Bucket=bucket,
        Key=key,
        Body=data,
        ContentType=contentType,
        Metadata=metadata
    )

# s3 - update bucket access ploicy
aws s3api put-public-access-block --bucket $mybucket --public-access-block-configuration "BlockPublicPolicy=false,RestrictPublicBuckets=false"

# s3 - sync local file to bucket
aws s3 sync ~/environment/labRepo/html/. s3://$mybucket/

# s3 - enable website host
aws s3api put-bucket-website --bucket $mybucket --website-configuration file://~/environment/labRepo/website.json

# s3 - update bucket policy
aws s3api put-bucket-policy --bucket $mybucket --policy file://~/environment/labRepo/policy.json
```
