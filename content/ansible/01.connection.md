#### Step2: Download && Install latest OpenSSL1.1.1 Archive

```bash
# open-ssl 1.1.1
cd /usr/local/src/
# Openssl package repository: https://www.openssl.org/source/
sudo wget https://www.openssl.org/source/openssl-1.1.1q.tar.gz
sudo tar -xzvf openssl-1.1.1q.tar.gz
cd openssl-1.1.1q
sudo ./config --prefix=/usr/local/ssl --openssldir=/usr/local/ssl shared zlib
sudo make
sudo make install
# configure openssl shared libraries
sudo touch /etc/ld.so.conf.d/openssl-1.1.1q.conf
echo '/usr/local/ssl/lib' | sudo tee -a /etc/ld.so.conf.d/openssl-1.1.1q.conf > /dev/null
# reload dynamic link
sudo ldconfig -v
# configure openssl binary
sudo mv /bin/openssl /bin/openssl.backup
sudo touch /etc/profile.d/openssl.sh
echo 'OPENSSL_PATH="/usr/local/ssl/bin"' | sudo tee -a /etc/profile.d/openssl.sh > /dev/null
echo 'export OPENSSL_PATH' | sudo tee -a /etc/profile.d/openssl.sh > /dev/null
echo 'PATH=$PATH:$OPENSSL_PATH' | sudo tee -a /etc/profile.d/openssl.sh > /dev/null
echo 'export PATH' | sudo tee -a /etc/profile.d/openssl.sh > /dev/null
# make file executable
sudo chmod +x /etc/profile.d/openssl.sh
# reload new openssl environment file and check the default PATH
source /etc/profile.d/openssl.sh
echo $PATH
# verify openssl installation
which openssl
openssl version -a
```

## 2. Create a new Windows server 2022 && Setup

### 2-1. Enable PowerShell Remoting for WinRm for Ansible

```powershell
Set-Service -Name "WinRM" -StartupType Automatic
Start-Service -Name "WinRM"
if (-not (Get-PSSessionConfiguration) -or (-not (Get-ChildItem WSMan:\localhost\Listener))) {
    ## Use SkipNetworkProfileCheck to make available even on Windows Firewall public profiles
    ## Use Force to not be prompted if we're sure or not.
    Enable-PSRemoting -SkipNetworkProfileCheck -Force
}
winrm enumerate winrm/config/Listener
```

### 2-2. Enable Certificate-Based Authentication

```powershell
Set-Item -Path WSMan:\localhost\Service\Auth\Certificate -Value $true
```

### 2-3. Create a Local User Account

```powershell
$UserAccountName = 'ansibleuser'
$UserAccountPassword = (ConvertTo-SecureString -String 'Password@123' -AsPlainText -Force)
if (-not (Get-LocalUser -Name $UserAccountName -ErrorAction Ignore)) {
    $newUserParams = @{
        Name                 = $UserAccountName
        AccountNeverExpires  = $true
        PasswordNeverExpires = $true
        Password             = $UserAccountPassword
    }
    $null = New-LocalUser @newUserParams
}
```

### 2-4. Import the public key into Trusted Root Certification Authorities and Trusted People

```powershell
# Modify certificate path
$pubKeyFilePath = 'C:\cert.pem'
$null = Import-Certificate -FilePath $pubKeyFilePath -CertStoreLocation 'Cert:\LocalMachine\Root'
$null = Import-Certificate -FilePath $pubKeyFilePath -CertStoreLocation 'Cert:\LocalMachine\TrustedPeople'
```

### 2-5. Create the Server Certificate

```powershell
# Modify <hostname>
$hostname = 'crdg'
$serverCert = New-SelfSignedCertificate -DnsName $hostName -CertStoreLocation 'Cert:\LocalMachine\My'
```

### 2-6. Create the Ansible WinRm Listener

```powershell
# Find all HTTPS listners
$httpsListeners = Get-ChildItem -Path WSMan:\localhost\Listener\ | where-object { $_.Keys -match 'Transport=HTTPS' }

# If not listeners are defined at all or no listener is configured to work with
# the server cert created, create a new one with a Subject of the computer's host name
# and bound to the server certificate.
if ((-not $httpsListeners) -or -not (@($httpsListeners).where( { $_.CertificateThumbprint -ne $serverCert.Thumbprint }))) {
    $newWsmanParams = @{
        ResourceUri = 'winrm/config/Listener'
        SelectorSet = @{ Transport = "HTTPS"; Address = "*" }
        ValueSet    = @{ Hostname = $hostName; CertificateThumbprint = $serverCert.Thumbprint }
        # UseSSL = $true
    }
    $null = New-WSManInstance @newWsmanParams
}
```

### 2-7. Map the Client Certificate to the Local User Account

```powershell
$credential = New-Object -TypeName System.Management.Automation.PSCredential -ArgumentList $UserAccountName, $UserAccountPassword

# Find the cert thumbprint for the client certificate created on the Ansible host
$ansibleCert = Get-ChildItem -Path 'Cert:\LocalMachine\Root' | Where-Object {$_.Subject -eq 'CN=ansibleuser'}

$params = @{
	Path = 'WSMan:\localhost\ClientCertificate'
	Subject = "$UserAccountName@localhost"
	URI = '*'
	Issuer = $ansibleCert.Thumbprint
  Credential = $credential
	Force = $true
}
New-Item @params
```

### 2-8. Allow WinRm for Ansible with User Account Control (UAC)

```powershell
$newItemParams = @{
    Path         = 'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Policies\System'
    Name         = 'LocalAccountTokenFilterPolicy'
    Value        = 1
    PropertyType = 'DWORD'
    Force        = $true
}
$null = New-ItemProperty @newItemParams
```

### 2-9. Open Port 5986 on the Windows Firewall

```powershell
# Ensure WinRM 5986 is open on the firewall
$ruleDisplayName = 'Windows Remote Management (HTTPS-In)'
if (-not (Get-NetFirewallRule -DisplayName $ruleDisplayName -ErrorAction Ignore)) {
    $newRuleParams = @{
        DisplayName   = $ruleDisplayName
        Direction     = 'Inbound'
        LocalPort     = 5986
        RemoteAddress = 'Any'
        Protocol      = 'TCP'
        Action        = 'Allow'
        Enabled       = 'True'
        Group         = 'Windows Remote Management'
    }
    $null = New-NetFirewallRule @newRuleParams
}
```

### 2-10. Add the Local User to the Administrators Group

```powershell
# Add the local user to the administrators group. If this step isn't doing, Ansible sees an "AccessDenied" error
Get-LocalUser -Name $UserAccountName | Add-LocalGroupMember -Group 'Administrators'
```
